วิธีเชื่อม Google Sheet "รวมคำศัพท์ IELTS":
  1. เปิดไฟล์ รวมคำศัพท์ IELTS ไปที่ Extensions > Apps Script
  2. วางโค้ดด้านล่างนี้ทับทั้งหมด กด Save แล้วกด Deploy > New deployment
  3. เลือกประเภทเป็น Web app ตั้งค่าสิทธิ์เป็น Anyone แล้วนำ URL มาวางในช่องด้านบน
const SHEET_ID = '1_emfLxKil9YRAcyOERRFiZDGr6zN6YRVbyaThRJ6PY0';
const TAB_NAME = 'Vocabulary';

function doGet(e) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName(TAB_NAME) || ss.getSheets()[0];
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0] || [];
  const items = rows.slice(1).filter(r => r[0]).map(r => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = r[i]);
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', items }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName(TAB_NAME) || ss.getSheets()[0];
  const body = JSON.parse(e.postData.contents || '{}');
  const items = Array.isArray(body.items) ? body.items : (body.item ? [body.item] : []);
  const data = sheet.getDataRange().getValues();
  const wordIndexMap = new Map();
  for (let i = 1; i < data.length; i++) {
    const w = String(data[i][0] || '').trim().toLowerCase();
    if (w) wordIndexMap.set(w, i + 1);
  }
  items.forEach(it => {
    const w = String(it.word || '').trim();
    if (!w) return;
    const key = w.toLowerCase();
    const rowVals = [
      w, it.pos || 'Word', it.phonetics || '', it.translation || it.meaning_th || '',
      it.context || '', Array.isArray(it.paraphrases) ? it.paraphrases.join(', ') : (it.paraphrases || ''),
      it.sourceApp || 'VocabPen', it.pdfName || '', it.page || '',
      it.status || 'Learning', new Date().toISOString()
    ];
    if (wordIndexMap.has(key)) {
      sheet.getRange(wordIndexMap.get(key), 1, 1, rowVals.length).setValues([rowVals]);
    } else {
      sheet.appendRow(rowVals);
      wordIndexMap.set(key, sheet.getLastRow());
    }
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: items.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
